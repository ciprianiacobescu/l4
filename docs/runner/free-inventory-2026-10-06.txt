## 7. The hello script used (scratch copy, run with mk-ssh.sh --script <file> -- <model>...)
    #!/usr/bin/env bash
    # free-hello.sh MODEL... — one tiny hello per model, as the runner user, in the L4 harness's opencode probe shape
    # (src/l4/agents/opencode.py `_hello`): empty cwd and XDG_CONFIG_HOME, the four OPENCODE_DISABLE_* switches, the
    # judge-mode OPENCODE_CONFIG_CONTENT (free-tier variant: read/bash "ask"), env -i with PATH/HOME/USER/LANG only,
    # argv `opencode run --pure --format json -m <model> --title "l4 hello" "Reply with exactly: L4-OK"`.
    # Prints the raw JSON stream between markers. Stops after the first 429 / rate-limit error.
    # MAGNITUDE=1 runs instead with the runner user's own config dir (the `magnitude` provider lives there), its MCP
    # servers disabled and the non-free judge config (tools off, permission deny) overlaid.
    set -u
    RUNNER_USER=gitlab-runner
    RHOME=$(getent passwd "$RUNNER_USER" | cut -d: -f6)
    PROMPT='Reply with exactly: L4-OK'
    FREE_CFG='{"$schema":"https://opencode.ai/config.json","snapshot":false,"share":"disabled","autoupdate":false,"permission":{"*":"deny","read":"ask","bash":"ask"}}'
    MAG_CFG='{"$schema":"https://opencode.ai/config.json","snapshot":false,"share":"disabled","autoupdate":false,"tools":{"*":false},"permission":{"*":"deny"},"mcp":{"firecrawl":{"type":"local","command":["true"],"enabled":false},"context7":{"type":"local","command":["true"],"enabled":false}}}'
    for model in "$@"; do
      echo "=== MODEL $model START $(date -u +%FT%TZ)"
      if [ "${MAGNITUDE:-0}" = 1 ]; then
        out=$(sudo -n -u "$RUNNER_USER" -H env -i HOME="$RHOME" USER="$RUNNER_USER" LANG=C.UTF-8 \
          PATH="$RHOME/.local/bin:/usr/local/bin:/usr/bin:/bin" OPENCODE_CONFIG_CONTENT="$MAG_CFG" \
          OPENCODE_DISABLE_PROJECT_CONFIG=1 OPENCODE_DISABLE_CLAUDE_CODE=1 OPENCODE_DISABLE_EXTERNAL_SKILLS=1 OPENCODE_DISABLE_AUTOUPDATE=1 \
          bash -c 'd=$(mktemp -d) && cd "$d" && s=$(date +%s%N); timeout 180 opencode run --pure --format json -m "$1" --title "l4 hello" "$2"; rc=$?; echo "__EXIT=$rc MS=$(( ($(date +%s%N)-s)/1000000 ))"; cd / && rm -rf "$d"' _ "$model" "$PROMPT" 2>&1)
      else
        out=$(sudo -n -u "$RUNNER_USER" -H env -i HOME="$RHOME" USER="$RUNNER_USER" LANG=C.UTF-8 \
          PATH="$RHOME/.local/bin:/usr/local/bin:/usr/bin:/bin" OPENCODE_CONFIG_CONTENT="$FREE_CFG" \
          OPENCODE_DISABLE_PROJECT_CONFIG=1 OPENCODE_DISABLE_CLAUDE_CODE=1 OPENCODE_DISABLE_EXTERNAL_SKILLS=1 OPENCODE_DISABLE_AUTOUPDATE=1 \
          bash -c 'd=$(mktemp -d) && mkdir "$d/cwd" "$d/config" && cd "$d/cwd" && s=$(date +%s%N); XDG_CONFIG_HOME="$d/config" timeout 180 opencode run --pure --format json -m "$1" --title "l4 hello" "$2"; rc=$?; echo "__EXIT=$rc MS=$(( ($(date +%s%N)-s)/1000000 ))"; cd / && rm -rf "$d"' _ "$model" "$PROMPT" 2>&1)
      fi
      printf '%s\n' "$out"
      echo "=== MODEL $model END"
      if printf '%s' "$out" | grep -q '"type":"error"' && printf '%s' "$out" | grep -q -i -E '429|rate.?limit|too many requests'; then
        echo "=== STOP: rate limit on $model"; break
      fi
    done
