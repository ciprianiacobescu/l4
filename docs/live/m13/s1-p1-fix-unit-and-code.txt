from decimal import ROUND_HALF_UP, Decimal


class UnsupportedShippingCountry(Exception):
    pass


def shipping_fee(subtotal: Decimal, country: str) -> Decimal:
    if subtotal < Decimal("0"):
        raise ValueError("subtotal cannot be negative")

    fees = {
        "RO": Decimal("4.99"),
        "DE": Decimal("9.99"),
    }
    if country not in fees:
        raise UnsupportedShippingCountry(country)

    fee = Decimal("0.00") if subtotal >= Decimal("50.00") else fees[country]
    return fee.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
